# Dependency-light functional harness, not the full engine or SDK build.
cmake_minimum_required(VERSION 3.20)
project(JudasPreAlphaWindowsChecks CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This harness validates native Windows/MSVC behaviour")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(cmake/JudasBuildIdentity.cmake)
judas_build_identity("${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /utf-8)
add_library(identity STATIC src/BuildIdentity.cpp src/BuildUpdate.cpp src/ToolProcess.cpp)
target_include_directories(identity PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(identity PRIVATE JUDAS_BUILD_CONFIGURATION="$<CONFIG>")
add_executable(judas_build_identity_tests tests/BuildIdentityTests.cpp)
target_link_libraries(judas_build_identity_tests PRIVATE identity)
add_executable(judas_network_transport_tests tests/NetworkTransportTests.cpp src/NetworkTransport.cpp)
target_include_directories(judas_network_transport_tests PRIVATE src)
target_link_libraries(judas_network_transport_tests PRIVATE ws2_32)
