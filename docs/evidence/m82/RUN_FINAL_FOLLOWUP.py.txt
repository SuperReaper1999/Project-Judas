from pathlib import Path
import os, subprocess, time, json, tempfile
ROOT=Path.cwd(); OUT=ROOT/'docs/evidence/m82/linux'
env=dict(os.environ, SDL_VIDEODRIVER='offscreen', SDL_AUDIODRIVER='dummy', XDG_DATA_HOME='/tmp/judas-m82-final-user')
results=[]
def run(label,args,timeout=180):
    started=time.monotonic()
    with (OUT/(label+'.txt')).open('w') as log:
        try: code=subprocess.run([str(x) for x in args],cwd=ROOT,env=env,stdout=log,stderr=subprocess.STDOUT,timeout=timeout).returncode
        except subprocess.TimeoutExpired: code=124
    results.append(dict(label=label,args=[str(x) for x in args],exit=code,seconds=time.monotonic()-started))
    print(label,code,flush=True)
    (OUT/'final-followup-cases.json').write_text(json.dumps(results,indent=2)+'\n')
    if code: raise SystemExit(code)
for name in ['powertrain','powertrain_lifecycle']:
    run(name,[ROOT/'build'/('judas_'+name+'_tests')])
script_root=Path(tempfile.mkdtemp(prefix='judas-m82-final-script-'))
run('powertrain_js',[ROOT/'build/judas_powertrain_script_tests',script_root])
run('api-live-coverage',['node',ROOT/'scripts/check_judasjs_api.mjs','--typescript','/tmp/judas-m80-typecheck/node_modules/typescript','--runtime',script_root/'surface.json'])
project=ROOT/'projects/powertrain_lab/powertrain_lab.judasproj'
for mode in ['lifecycle','write','read']:
    run('application-'+mode,[ROOT/'build/judas_powertrain_application_tests',project,mode,OUT/('application-'+mode)])
wheel=ROOT/'projects/wheel_lab/wheel_lab.judasproj'
for mode in ['probe','oblique','radial','lifecycle']:
    run('wheel-application-'+mode,[ROOT/'build/judas_wheel_application_tests',wheel,mode,OUT/('wheel-application-'+mode)])
stream=ROOT/'projects/streamed_range/streamed_range.judasproj'
for mode in ['write','read']:
    run('regression-stream-'+mode,[ROOT/'build/judas_save_streaming_tests',stream,mode,OUT/('regression-stream-'+mode)],300)
