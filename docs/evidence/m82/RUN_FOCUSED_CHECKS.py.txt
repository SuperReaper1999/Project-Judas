from pathlib import Path
import os, subprocess, time, json
ROOT=Path.cwd()
OUT=ROOT/'docs/evidence/m82/linux'
env=dict(os.environ,SDL_VIDEODRIVER='offscreen',SDL_AUDIODRIVER='dummy',XDG_DATA_HOME='/tmp/judas-m82-final-user')
results=[]
def run(label,args,timeout=180):
    path=OUT/(label+'.txt'); start=time.monotonic()
    with path.open('w') as log:
        try: code=subprocess.run([str(x) for x in args],cwd=ROOT,env=env,stdout=log,stderr=subprocess.STDOUT,timeout=timeout).returncode
        except subprocess.TimeoutExpired: code=124
    results.append(dict(label=label,args=[str(x) for x in args],exit=code,seconds=time.monotonic()-start))
    print(label,code,flush=True)
    (OUT/'focused-cases.json').write_text(json.dumps(results,indent=2)+'\n')
    return code
for name in ['powertrain','powertrain_lifecycle','wheel','wheel_lifecycle','physics','collision','physics_cast','joint','joint_authoring','character_motor','kinematic','physics_continuation','save_storage','bending','structural','gravity_selection','uniform_gravity_scene']:
    run(name,[ROOT/'build'/('judas_'+name+'_tests')])
run('save-exclusion',[ROOT/'build/judas_save_exclusion_tests','/tmp/judas-m82-save-exclusion-fresh',ROOT])
run('powertrain_js',[ROOT/'build/judas_powertrain_script_tests','/tmp/judas-m82-final-script'])
run('wheel_js',[ROOT/'build/judas_wheel_script_tests','/tmp/judas-m82-wheel-script'])
project=ROOT/'projects/powertrain_lab/powertrain_lab.judasproj'
for mode in ['flat','oblique','radial','contracts','lifecycle','write','read']:
    run('application-'+mode,[ROOT/'build/judas_powertrain_application_tests',project,mode,OUT/('application-'+mode)])
wheel=ROOT/'projects/wheel_lab/wheel_lab.judasproj'
for mode in ['probe','oblique','radial','lifecycle']:
    run('wheel-application-'+mode,[ROOT/'build/judas_wheel_application_tests',wheel,mode,OUT/('wheel-application-'+mode)])
stream=ROOT/'projects/streamed_range/streamed_range.judasproj'
for mode in ['write','read']:
    run('regression-stream-'+mode,[ROOT/'build/judas_save_streaming_tests',stream,mode,OUT/('regression-stream-'+mode)],300)
raise SystemExit(any(r['exit'] for r in results))
