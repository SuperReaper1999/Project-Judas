#include "Deformable.h"
#include "UniformGravity.h"
#include <cstdio>
int main(){auto a=std::make_shared<DeformableAsset>(MakeDeformableBlock({4,2,2},{3,1,1}));PhysicsWorld p;p.Init();UniformGravity g({0,-9.81f,0});DeformableTarget t;t.valid=true;for(double yield:{0.,.04,.06,.08}){DeformableSettings c;c.selfContact=true;c.material.density=1;c.material.shearCompliance=.0005;c.material.volumeCompliance=.000005;c.material.damping=1;c.material.yieldStrain=yield;c.material.plasticRate=1;DeformableAttachment pin;pin.group="left";c.attachments={pin};DeformableInstance s;s.Initialize(a,c,glm::dmat4(1),1);double calm=0,loaded=0,restCalm=0,restFinal=0;for(int i=0;i<480;++i){if(i>=120&&i<240)s.Force("right",{0,-40,0});s.Step(1./60,p,g,{t});if(i<120)calm=std::max(calm,s.stats.maximumStrain);else if(i<240)loaded=std::max(loaded,s.stats.maximumStrain);double rest=0;for(unsigned j=0;j<a->solids.size();++j)for(int k=0;k<3;++k)rest=std::max(rest,glm::length(s.plasticRest[j][k]-a->solids[j].rest[k]));if(i==119)restCalm=rest;restFinal=rest;}std::printf("yield=%g calmPeak=%g loadPeak=%g calmRest=%g finalRest=%g error=%s\n",yield,calm,loaded,restCalm,restFinal,s.error.c_str());}}
