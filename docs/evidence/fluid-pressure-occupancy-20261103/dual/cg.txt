    // Solve a search direction on the currently free dual rows. Rebuild that
    // face when a multiplier reaches zero or an inactive row violates.
    std::vector<double> residual(rows.size()), direction(rows.size()), previous(rows.size()), q(rows.size());
    std::vector<bool> active(rows.size()), previousActive(rows.size());
    std::vector<glm::dvec3> transpose(count);
    double previousSquared=0;
    bool restart=true;
    const auto measure=[&]() {
        result.residual=0;
        for(std::size_t j=0;j<rows.size();++j) {
            residual[j]=error(rows[j]);
            result.residual=std::max(result.residual,lambda[j]>0?std::abs(residual[j]):residual[j]);
        }
    };
    // Each outer iteration applies the normalized dual Hessian exactly once.
    measure();
    for(;result.sweeps<maxSweeps && result.residual>tolerance;++result.sweeps) {
        double squared=0;
        for(std::size_t j=0;j<rows.size();++j) {
            active[j]=lambda[j]>0 || residual[j]>tolerance;
            if(active[j])squared+=residual[j]*residual[j];
        }
        if(!(squared>0) || !std::isfinite(squared))break;
        restart=restart || active!=previousActive;
        const double beta=restart?0:squared/previousSquared;
        std::fill(transpose.begin(),transpose.end(),glm::dvec3(0));
        for(std::size_t j=0;j<rows.size();++j) {
            direction[j]=active[j]?residual[j]+beta*previous[j]:0;
            const auto& row=rows[j];
            if(direction[j]==0)continue;
            for(std::size_t e=row.begin;e<row.end;++e)
                transpose[coefficients[e].particle]+=direction[j]*coefficients[e].gradient;
        }
        double curvature=0, descent=0;
        for(const auto& value:transpose)curvature+=glm::dot(value,value);
        for(std::size_t j=0;j<rows.size();++j)descent+=direction[j]*residual[j];
        double alpha=descent/curvature;
        std::size_t hit=rows.size();
        if(alpha>0 && std::isfinite(alpha)) {
            for(std::size_t j=0;j<rows.size();++j) {
                if(direction[j]>=0)continue;
                const double limit=-lambda[j]/direction[j];
                if(limit<alpha){alpha=limit;hit=j;}
            }
        }
        if(!(alpha>0) || !std::isfinite(alpha)) {
            // A singular face or roundoff-blocked CG direction cannot stop
            // progress. One full projected Gauss-Seidel sweep is a safe fallback.
            for(std::size_t j=0;j<rows.size();++j) {
                const auto& row=rows[j];
                const double next=std::max(0.0,lambda[j]+error(row));
                const double change=next-lambda[j];
                if(change==0)continue;
                lambda[j]=next;
                for(std::size_t e=row.begin;e<row.end;++e)
                    solution[coefficients[e].particle]-=change*coefficients[e].gradient;
            }
            restart=true;
        } else {
            for(std::size_t j=0;j<rows.size();++j) {
                lambda[j]=std::max(0.,lambda[j]+alpha*direction[j]);
                if(j==hit)lambda[j]=0;
            }
            for(std::size_t i=0;i<count;++i)solution[i]-=alpha*transpose[i];
            restart=hit<rows.size();
        }
        previousActive=active;
        previous=direction;
        previousSquared=squared;
        measure();
    }
